$sparql = @'
SELECT DISTINCT ?obj ?objLabel ?file ?url (YEAR(?date) AS ?year) WHERE {
  VALUES ?mainObj {
    wd:Q113584 wd:Q158148 wd:Q155422 wd:Q151130 wd:Q154045 # OTMAA
  }
  ?file wdt:P180 ?mainObj.
  ?file wdt:P180 ?obj.
  OPTIONAL { ?file wdt:P571 ?date }
  
  # Only keep the main objects or these two
  FILTER(?obj = ?mainObj || ?obj IN (wd:Q40787, wd:Q152094))
  
  ?file schema:contentUrl ?url.
  FILTER(NOT EXISTS { ?file wdt:P136 wd:Q482919. })
  
  # Optional label for the depicted object
  SERVICE <https://query.wikidata.org/sparql> {
    OPTIONAL {
      ?obj rdfs:label ?objLabel.
      FILTER((LANG(?objLabel)) = "ru")
    }
  }
}
'@
$resp = Invoke-Sparql $sparql 'https://commons-query.wikimedia.org/sparql'
$objs = $resp | sort {$_.year -eq '' ? [int]$_.year : $null} | group file | % {
    $url = $_.group[0].url
    $encfilename = $url -replace '.*/'
    $urlThumb = "https://commons.wikimedia.org/w/index.php?title=Special:Redirect/file/$encfilename&width=200"
    [array]$tags = $_.group.objLabel | select -Unique
    [pscustomobject]@{url=$urlThumb; action=$_.group[0].file; depicted=$tags;}
}
$objs | Format-HtmlGallery -Description $descr -Title "OTMAA full" -TagsProperty depicted
