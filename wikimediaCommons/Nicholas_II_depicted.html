# WCQS: With whom Nicholas II is depicted
SELECT ?obj ?objLabel ?objDesc (COUNT(DISTINCT ?file) AS ?depictedBothCount) ?depictedBoth ?depicted ?categ ?inCategWoDepicts WHERE { 
  VALUES ?target {wd:Q40787}
  ?file wdt:P180 ?target .
  ?file wdt:P180 ?obj .
  
  #timeout (COUNT(DISTINCT ?fileWithObj) AS ?depictedItself)
  #?fileWithObj wdt:P180 ?obj .
  
  BIND(REPLACE(STR(?target), ".*/", "") AS ?targetQid)
  # BIND(REPLACE(STR(?obj), ".*/", "") AS ?qid)
  # Construct the URL
  BIND(URI(CONCAT("https://commons.wikimedia.org/w/index.php?search=haswbstatement%3AP180%3D",?targetQid,"+haswbstatement%3AP180%3D", ?qid, "&title=Special:MediaSearch&type=image")) AS ?depictedBoth)
  BIND(URI(CONCAT("https://commons.wikimedia.org/w/index.php?search=haswbstatement%3AP180%3D",?qid, "&title=Special:MediaSearch&type=image")) AS ?depicted)
  SERVICE <https://query.wikidata.org/sparql> {
    ?obj wdt:P31 wd:Q5 .
    OPTIONAL {
      ?obj rdfs:label ?objLabel .
      FILTER (lang(?objLabel) = 'ru') 
    }
    OPTIONAL {
    ?obj schema:description ?objDesc .
    FILTER ((LANG(?objDesc)) = "ru")
  }
  OPTIONAL { 
    ?obj wdt:P373 ?categTitle . 
    BIND(uri(CONCAT("https://commons.wikimedia.org/wiki/Category:",?categTitle)) AS ?categ)
    BIND(REPLACE(STR(?obj), ".*/", "") AS ?qid)
    BIND(ENCODE_FOR_URI(CONCAT("deepcategory:\"",?categTitle,"\" -haswbstatement:P180=",?qid)) AS ?query)
    BIND(uri(CONCAT("https://commons.wikimedia.org/w/index.php?search=",?query,"&title=Special:MediaSearch&type=image")) AS ?inCategWoDepicts) 
  }
  }
}
GROUP BY ?obj ?objLabel ?objDesc ?depicted ?depictedBoth ?categ ?inCategWoDepicts 
ORDER BY DESC (?depictedBothCount)
