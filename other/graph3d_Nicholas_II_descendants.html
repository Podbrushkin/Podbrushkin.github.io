$sparql =@'
#defaultView:Graph
# All Nicholas I descendants
PREFIX gas: <http://www.bigdata.com/rdf/gas#>

SELECT ?item ?itemLabel (SAMPLE(YEAR(?birthDate)) AS ?birthYear) (SAMPLE(YEAR(?deathDate)) AS ?deathYear) ?links ?parentId
WHERE {
  SERVICE gas:service {
    gas:program gas:gasClass "com.bigdata.rdf.graph.analytics.SSSP" ;
                gas:in wd:Q130734 ;
                gas:traversalDirection "Forward" ;
                gas:out ?item ;
                gas:out1 ?depth ;
                gas:out2 ?parentId ;  # Capture the parent ID
                gas:maxIterations 4 ;
                gas:linkType wdt:P40 .
  }
  OPTIONAL { ?item wdt:P569 ?birthDate. }
  OPTIONAL { ?item wdt:P570 ?deathDate. }
  OPTIONAL { ?item wikibase:sitelinks ?links. }
  #OPTIONAL { ?item wdt:P40 ?linkTo }
  #OPTIONAL { ?item wdt:P18 ?pic }
  SERVICE wikibase:label { bd:serviceParam wikibase:language "ru,en" }
}
GROUP BY ?item ?itemLabel ?links ?parentId
'@
$ppl = Invoke-Sparql $sparql
$ppl | % {
  Add-Member -inp $_ -NotePropertyName label -NotePropertyValue "$($_.itemLabel) ($($_.birthYear)-$($_.deathYear))"
}
$nodes = $ppl | select @{n='id';e='item'},label,@{n='value';e='links'},@{n='url';e='item'}
$edges = $ppl | ? parentId | select @{n='source';e='parentId'},@{n='target';e='item'}
Format-HtmlGraph3D -nodes $nodes -edges $edges -OutVariable html | Invoke-Item
