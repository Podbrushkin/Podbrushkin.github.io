# get subcategories
$url = 'https://petscan.wmflabs.org/'
$resp = irm $url -Body @{doit=1; language='ru'; project='wikisource'; categories='Индекс'; depth=0; format='json';}

#for each subcategory get all children
$resp.'*'[0].a.'*' | ? namespace -eq 14 | % title | % {
$category = $_
$url = 'https://petscan.wmflabs.org/'
$resp = irm $url -Body @{doit=1; language='ru'; project='wikisource'; categories=$_; depth=0; format='json';}
try {
$resp.'*'[0].a.'*' | % {Add-Member -inp $_ -NotePropertyName category -NotePropertyValue $category; $_}
} catch {Write-Host $_}
} -OutVariable indices
