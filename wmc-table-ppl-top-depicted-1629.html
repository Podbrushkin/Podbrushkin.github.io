# For WIKIDATA endpoint, most depicted people on Commons
SELECT ?obj ?objLabel ?born ?sex ?count ?links ?url ?categ WITH {
	SELECT ?count ?obj  WHERE {
		SERVICE <https://qlever.cs.uni-freiburg.de/api/wikimedia-commons> {
			SELECT (COUNT(DISTINCT ?file) AS ?count) ?obj WHERE {
				?file wdt:P180 ?obj .
			}
			GROUP BY ?obj
			HAVING(?count > 200)
		}
	}
} AS %items
WHERE {
INCLUDE %items
	 ?obj wdt:P31 wd:Q5 .
  BIND(REPLACE(STR(?obj), ".*/", "") AS ?qid)
  # Construct the URL
  BIND(URI(CONCAT("https://commons.wikimedia.org/w/index.php?search=haswbstatement%3AP180%3D", ?qid, "&title=Special:MediaSearch&type=image")) AS ?url)
  OPTIONAL {?obj wdt:P21 ?sex .}
  OPTIONAL {?obj wdt:P569 ?born .}
  
  OPTIONAL {
    ?obj wdt:P373 ?categTitle .
    BIND(uri(CONCAT("https://commons.wikimedia.org/wiki/Category:",?categTitle)) AS ?categ)
  }
  
  SERVICE wikibase:label {  bd:serviceParam wikibase:language "ru,en". }
  OPTIONAL {?obj wikibase:sitelinks ?links }
}
ORDER BY DESC(?count)
