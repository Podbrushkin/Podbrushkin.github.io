$sparql = @'
#efaultView:Graph
PREFIX gas: <http://www.bigdata.com/rdf/gas#>

SELECT ?obj ?objLabel ?links ?parent
WHERE {
  SERVICE gas:service {
    gas:program gas:gasClass "com.bigdata.rdf.graph.analytics.SSSP" ;
                gas:in wd:Q19088 ;
                gas:traversalDirection "Reverse" ;
                gas:out ?obj ;
                #gas:out1 ?depth ;
                gas:out2 ?parent ;  # Capture the parent ID
                gas:maxIterations 2 ;
                gas:linkType wdt:P171 .
  }
  ?obj wikibase:sitelinks ?links.
  SERVICE wikibase:label { bd:serviceParam wikibase:language "ru,mul,en" }
}
'@
$taxons = Invoke-Sparql $sparql

$nodes = $taxons | select @{n='id';e='obj'},@{n='label';e='objLabel'},@{n='url';e={$_.wikipedia ?? $_.obj}},@{n='value';e='links'}
$edges = $taxons | ? {$_.parent -In $nodes.id} | select @{n='source';e='parent'},@{n='target';e='obj'}
Format-HtmlGraph -nodes $nodes -edges $edges -OutVariable html | Invoke-Item
