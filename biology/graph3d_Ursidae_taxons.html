$sparql = @'
SELECT DISTINCT ?obj ?objLabel ?links ?wikipedia ?parent WHERE {
  ?obj (wdt:P171*) wd:Q11788.
  ?obj  wdt:P171 ?parent.
  ?obj  wikibase:sitelinks ?links.
  OPTIONAL {
    ?wikipedia schema:about ?obj;
      schema:isPartOf <https://ru.wikipedia.org/>.
  }
  SERVICE wikibase:label { bd:serviceParam wikibase:language "ru,mul,en". }
}
ORDER BY DESC(?links)
'@
$taxons = Invoke-Sparql $sparql
$taxons | ? wikipedia -Match '^$' | % {$_.wikipedia = $null}
$nodes = $taxons | select @{n='id';e='obj'},@{n='label';e='objLabel'},@{n='url';e={$_.wikipedia ?? $_.obj}},@{n='value';e='links'}
$edges = $taxons | ? {$_.parent -In $nodes.id} | select @{n='source';e='parent'},@{n='target';e='obj'}
Format-HtmlGraph3D -nodes $nodes -edges $edges -OutVariable html | Invoke-Item
